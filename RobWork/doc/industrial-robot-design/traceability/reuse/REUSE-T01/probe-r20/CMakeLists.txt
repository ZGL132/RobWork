cmake_minimum_required(VERSION 3.23)
project(ird_project_boundary_audit LANGUAGES CXX)
# 复用原project/core/testkit目标与原事务夹具，不复制生产源集或新建事务引擎。
find_package(RobWork REQUIRED CONFIG COMPONENTS common models)
find_package(GTest CONFIG REQUIRED)
foreach(component math kinematics models)
    if(NOT TARGET sdurw_${component})
        add_library(sdurw_${component} ALIAS RW::sdurw_${component})
    endif()
endforeach()
get_filename_component(IRD_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../RobWorkStudio/src/rwslibs/industrialrobot" ABSOLUTE)
add_subdirectory("${IRD_SOURCE_DIR}/core" "${CMAKE_CURRENT_BINARY_DIR}/ird-core" EXCLUDE_FROM_ALL)
add_subdirectory("${IRD_SOURCE_DIR}/testkit" "${CMAKE_CURRENT_BINARY_DIR}/ird-testkit" EXCLUDE_FROM_ALL)
add_subdirectory("${IRD_SOURCE_DIR}/project" "${CMAKE_CURRENT_BINARY_DIR}/ird-project" EXCLUDE_FROM_ALL)
add_executable(ird_project_boundary_audit ProjectBoundaryTest.cpp "${IRD_SOURCE_DIR}/core/test/TestMainReport.cpp")
set_target_properties(ird_project_boundary_audit PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
if(MSVC)
    target_compile_options(ird_project_boundary_audit PRIVATE /utf-8)
endif()
target_include_directories(ird_project_boundary_audit PRIVATE "${IRD_SOURCE_DIR}/project/src")
target_compile_definitions(ird_project_boundary_audit PRIVATE IRD_PROJECT_EXISTING_TEST="${IRD_SOURCE_DIR}/project/test/TxEngineTest.cpp")
target_link_libraries(ird_project_boundary_audit PRIVATE sdurws_ird_project sdurws_ird_testkit GTest::gtest)
