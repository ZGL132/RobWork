cmake_minimum_required(VERSION 3.23)
project(ird_execution_boundary_audit LANGUAGES CXX)
# 审计只接原目标、原登记/接纳夹具与原报告入口，不复制生产源集。
find_package(RobWork REQUIRED CONFIG COMPONENTS common models)
find_package(GTest CONFIG REQUIRED)
foreach(component math kinematics models)
    if(NOT TARGET sdurw_${component})
        add_library(sdurw_${component} ALIAS RW::sdurw_${component})
    endif()
endforeach()
if(NOT IRD_SOURCE_DIR)
    message(FATAL_ERROR "IRD_SOURCE_DIR must point to the inspected industrialrobot source")
endif()
foreach(unit core testkit evidence project execution)
    add_subdirectory("${IRD_SOURCE_DIR}/${unit}" "${CMAKE_CURRENT_BINARY_DIR}/ird-${unit}" EXCLUDE_FROM_ALL)
endforeach()
add_executable(ird_execution_boundary_audit RegistryBoundaryTest.cpp "${IRD_SOURCE_DIR}/execution/test/TestMainReport.cpp")
set_target_properties(ird_execution_boundary_audit PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
if(MSVC)
    target_compile_options(ird_execution_boundary_audit PRIVATE /utf-8)
endif()
target_compile_definitions(ird_execution_boundary_audit PRIVATE IRD_REGISTRY_EXISTING_TEST="${IRD_SOURCE_DIR}/execution/test/RunRegistryAdmissionTest.cpp")
target_link_libraries(ird_execution_boundary_audit PRIVATE sdurws_ird_execution sdurws_ird_testkit GTest::gtest)
