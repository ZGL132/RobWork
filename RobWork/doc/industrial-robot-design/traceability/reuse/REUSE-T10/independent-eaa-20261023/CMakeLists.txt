cmake_minimum_required(VERSION 3.16)
project(EaaIndependentBoundary LANGUAGES CXX)
find_package(RobWork CONFIG REQUIRED)
find_package(GTest CONFIG REQUIRED)
add_executable(eaa_boundary Boundary.cpp)
set_target_properties(eaa_boundary PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_options(eaa_boundary PRIVATE /utf-8)
target_compile_definitions(eaa_boundary PRIVATE EIGEN_DONT_ALIGN_STATICALLY=1)
target_link_libraries(eaa_boundary PRIVATE RW::sdurw_math GTest::gtest_main)
add_executable(eaa_diagnostic Diagnostic.cpp)
set_target_properties(eaa_diagnostic PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_options(eaa_diagnostic PRIVATE /utf-8)
target_compile_definitions(eaa_diagnostic PRIVATE EIGEN_DONT_ALIGN_STATICALLY=1 NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(eaa_diagnostic PRIVATE RW::sdurw_math)
