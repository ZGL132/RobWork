cmake_minimum_required(VERSION 3.16)
project(EaaOriginalEnvelope LANGUAGES CXX)
find_package(RobWork CONFIG REQUIRED)
find_package(GTest CONFIG REQUIRED)
add_executable(eaa_original_envelope Envelope.cpp)
set_target_properties(eaa_original_envelope PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_options(eaa_original_envelope PRIVATE /utf-8)
target_compile_definitions(eaa_original_envelope PRIVATE EIGEN_DONT_ALIGN_STATICALLY=1)
target_link_libraries(eaa_original_envelope PRIVATE RW::sdurw_math GTest::gtest_main)
