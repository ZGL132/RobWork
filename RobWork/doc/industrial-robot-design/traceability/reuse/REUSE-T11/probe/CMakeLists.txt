cmake_minimum_required(VERSION 3.16)
project(RpyReuseProbe LANGUAGES CXX)
find_package(RobWork CONFIG REQUIRED)
find_package(GTest CONFIG REQUIRED)
# 复用既有数学库/GTest；同一消费者源码分别对比原薄适配和重构版本，不复制产品构建。
foreach(mode IN ITEMS current baseline)
  add_executable(rpy_${mode} Probe.cpp)
  set_target_properties(rpy_${mode} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  target_compile_options(rpy_${mode} PRIVATE /utf-8)
  target_compile_definitions(rpy_${mode} PRIVATE EIGEN_DONT_ALIGN_STATICALLY=1)
  target_link_libraries(rpy_${mode} PRIVATE RW::sdurw_math GTest::gtest_main)
endforeach()
target_include_directories(rpy_current PRIVATE "${PRODUCT_SOURCE_DIR}/modeling/src")
target_include_directories(rpy_baseline PRIVATE "${BASELINE_HEADER_DIR}")
