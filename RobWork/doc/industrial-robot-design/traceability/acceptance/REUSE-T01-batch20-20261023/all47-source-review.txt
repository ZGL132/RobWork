261: std::string TxEngine::generateTxId()
262: {
263:     // §4.1 .staging 行："<tx-id>＝进程内单调＋随机后缀"。单调序号防同
264:     // 进程内重复；随机后缀防跨进程/跨会话残留目录撞名（残留会被恢复

276:             hex[i * 8 + j] = kDigits[(v >> (j * 4)) & 0xF];
277:         }
278:     }
279:     hex[16] = '\0';
280:     return "tx-" + std::to_string(seq) + "-" + std::string{hex, 16};
