+            }
+            if (meshes != (text[0]=='s' ? 0u : 1u)) { return "实际网格数不一致："+std::to_string(meshes); }
+          }
+        }
+        return meshDirectories()==directoriesBefore ? "" : "加载成功/异常后遗留临时目录";
+    });
