FUNCTION runtime.symmetrizedSpdWitness RobWork/RobWorkStudio/src/rwslibs/industrialrobot/runtime/src/CanonicalModel.cpp 85-134
85: double symmetrizedSpdWitness(const rw::math::InertiaMatrix<double>& m) noexcept
86: {
87:     // 第一步：SPD 必须有严格正对角。用二进制幂对每轴做合同变换 D⁻¹ S D⁻¹，
88:     // S=(I+Iᵀ)/2；正定惯性不变。按轴平衡而非全矩阵除以最大值，避免
89:     // diag(denorm_min,1,1e300) 的小正轴被全局缩放吞成零。
90:     int power[3]{};
91:     for (int i = 0; i < 3; ++i) {
92:         if (!(m(i, i) > 0.0)) { return m(i, i) == 0.0 ? 0.0 : -1.0; }
93:         int exponent = 0;
94:         std::frexp(m(i, i), &exponent);
95:         // floor(exponent/2)，显式处理 C++ 负整数除法向零截断。
96:         power[i] = exponent >= 0 ? exponent / 2 : (exponent - 1) / 2;
97:     }
98:     double s[3][3]{};
99:     for (int i = 0; i < 3; ++i) {
100:         for (int j = i; j < 3; ++j) {
101:             const double a = m(i, j);
102:             const double b = m(j, i);
103:             // 同号用差值平均防 a+b 溢出，异号先各减半防 b-a 溢出；
104:             // 对角不变，最小正次正规数不会因先除以 2 而消失。
105:             const double average = std::signbit(a) == std::signbit(b)
106:                 ? a + 0.5 * (b - a) : 0.5 * a + 0.5 * b;
107:             s[i][j] = s[j][i] = std::scalbn(average, -power[i] - power[j]);
108:             if (!std::isfinite(s[i][j])) {
109:                 // 平衡后对角在 [0.5,2)，交叉项若溢出，必违背 SPD 的
110:                 // |Sij|²<Sii*Sjj 必要条件；拒绝而不夹紧成合法矩阵。
111:                 return -1.0;
112:             }
113:             // 平衡后的对角均小于 2，|交叉项|>2 必不满足 Cauchy 必要界；
114:             // 在后续求平方前拒绝，确保极大有限非法输入也不会产生 NaN 主元。
115:             if (std::fabs(s[i][j]) > 2.0) { return -1.0; }
116:         }
117:     }
118: 
119:     // 第二步：固定顺序 LDLᵀ。二阶主子式用两次 FMA 补偿乘积舍入，
120:     // 防对称化后恰为零的主子式被加减舍入成小正数而误放行。
121:     const double square = s[0][1] * s[0][1];
122:     const double determinant2 = std::fma(s[0][0], s[1][1], -square)
123:         - std::fma(s[0][1], s[0][1], -square);
124:     const double pivot2 = determinant2 / s[0][0];
125:     if (!(pivot2 > 0.0)) { return pivot2; } // 零=半正定，负=不定，均须拒绝。
126: 
127:     // 第三步：继续消去第一、第二轴；仅两两主子式为正不能保证三阶 SPD。
128:     const double schur22 = std::fma(-s[0][2] / s[0][0], s[0][2], s[2][2]);
129:     const double schur12 = std::fma(-s[0][1] / s[0][0], s[0][2], s[1][2]);
130:     const double pivot3 = std::fma(-schur12 / pivot2, schur12, schur22);
131:     // 极病态非法矩阵的除法可能溢出；非有限见证一律拒绝，诊断值保持有限。
132:     if (!std::isfinite(pivot3)) { return -1.0; }
133:     return std::min({s[0][0], pivot2, pivot3});
134: }
FUNCTION runtime.relativeAsymmetry RobWork/RobWorkStudio/src/rwslibs/industrialrobot/runtime/src/CanonicalModel.cpp 50-73
50: double relativeAsymmetry(const rw::math::InertiaMatrix<double>& m) noexcept
51: {
52:     // 第一步：共同除以最大绝对分量后再取范数，避免 I² 上溢/下溢。公因子
53:     // 在比值中消去，不把接近零的惯量当成零，也不引入额外绝对容差。
54:     double scale = 0.0;
55:     for (int i = 0; i < 3; ++i) {
56:         for (int j = 0; j < 3; ++j) { scale = std::max(scale, std::fabs(m(i, j))); }
57:     }
58:     if (scale == 0.0) { return 0.0; } // 零张量的对称性成立，随后严格 SPD 会拒绝。
59:     double symNorm = 0.0;
60:     double worst = 0.0;
61:     for (int i = 0; i < 3; ++i) {
62:         for (int j = 0; j < 3; ++j) {
63:             const double a = m(i, j) / scale;
64:             const double b = m(j, i) / scale;
65:             symNorm = std::hypot(symNorm, 0.5 * a + 0.5 * b);
66:             worst = std::max(worst, std::fabs(a - b));
67:         }
68:     }
69:     // 第二步：分母只取物理对称部分，不能让反对称误差放大自己的容许上限。
70:     // 纯反对称时退回绝对差（dynamics 已有零参考规则），之后仍须通过 SPD。
71:     const double deviation = symNorm > 0.0 ? worst / symNorm : worst * scale;
72:     return std::min(deviation, std::numeric_limits<double>::max());
73: }
