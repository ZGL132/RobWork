133: std::optional<CollisionDomain> classifyDomainPair(SceneObjectRole a, SceneObjectRole b)
134: {
137:     if (static_cast<int>(b) < static_cast<int>(a)) {
138:         std::swap(a, b);
139:     }
141:     switch (a) {
142:     case SceneObjectRole::RobotLink:
143:         switch (b) {
144:         case SceneObjectRole::RobotLink:         return CollisionDomain::Self;        // 机器人非相邻/相邻连杆
145:         case SceneObjectRole::Tool:              return CollisionDomain::Tool;        // 工具 vs 连杆（安装侧相邻/非相邻）
146:         case SceneObjectRole::Payload:           return CollisionDomain::Tool;        // 负载 vs 连杆
147:         case SceneObjectRole::EnvironmentObject: return CollisionDomain::Environment; // 连杆 vs 环境对象
148:         case SceneObjectRole::Workpiece:         return CollisionDomain::Environment; // 连杆 vs 工件（工件归环境侧）
149:         }
150:         break;
151:     case SceneObjectRole::Tool:
152:         switch (b) {
153:         case SceneObjectRole::Tool:              return std::nullopt;                 // 工具×工具——矩阵未列
154:         case SceneObjectRole::Payload:           return std::nullopt;                 // 工具×负载——矩阵未列
155:         case SceneObjectRole::EnvironmentObject: return CollisionDomain::Environment; // 机器人侧 vs 环境对象
156:         case SceneObjectRole::Workpiece:         return CollisionDomain::Tool;        // 工具 vs 工件（Tool 域显式行）
157:         }
158:         break;
159:     case SceneObjectRole::Payload:
160:         switch (b) {
161:         case SceneObjectRole::Payload:           return std::nullopt;                 // 负载×负载——矩阵未列
162:         case SceneObjectRole::EnvironmentObject: return CollisionDomain::Environment; // 机器人侧 vs 环境对象
163:         case SceneObjectRole::Workpiece:         return CollisionDomain::Tool;        // 负载 vs 工件
164:         }

196: core::ContentIdentity computeSessionIdentity(const core::ContentIdentity& policyCid,

220:             static_cast<std::uint8_t>(part->size() & 0xFFu)};

508: void expandRulesAndCheckConflicts(const EngineeringPolicySet& policy,
509:                                   const SceneObjectIndex& objectIndex,
510:                                   std::map<PairKey, std::size_t>& mandatoryExpanded,
511:                                   std::map<PairKey, std::pair<std::size_t, std::string>>&
512:                                       excludedExpanded)
513: {
515:     for (std::size_t i = 0; i < policy.collision.mandatoryPairs.size(); ++i) {
516:         const PairRule& rule = policy.collision.mandatoryPairs[i];
517:         const std::vector<core::ObjectId> first = expandRuleTarget(rule.first, objectIndex);
518:         const std::vector<core::ObjectId> second = expandRuleTarget(rule.second, objectIndex);
519:         for (const core::ObjectId& a : first) {
520:             for (const core::ObjectId& b : second) {
521:                 if (a == b) {
522:                     continue;   // 自对无碰撞语义（同一对象不与自己检测）
523:                 }
524:                 mandatoryExpanded.emplace(detail::canonicalPair(a, b), i);
525:             }
526:         }
527:     }
529:     for (std::size_t i = 0; i < policy.collision.excludedPairs.size(); ++i) {
530:         const PairRule& rule = policy.collision.excludedPairs[i];
531:         const std::vector<core::ObjectId> first = expandRuleTarget(rule.first, objectIndex);
532:         const std::vector<core::ObjectId> second = expandRuleTarget(rule.second, objectIndex);
533:         for (const core::ObjectId& a : first) {
534:             for (const core::ObjectId& b : second) {
535:                 if (a == b) {
536:                     continue;
537:                 }
538:                 excludedExpanded.emplace(detail::canonicalPair(a, b),
539:                                          std::make_pair(i, rule.reason));
540:             }
541:         }
542:     }
546:     for (const auto& [pairKey, mandatoryIndex] : mandatoryExpanded) {
547:         const auto excludedIt = excludedExpanded.find(pairKey);
548:         if (excludedIt != excludedExpanded.end()) {
549:             throw PolicyError(
550:                 PolicyErrorCode::RuleConflict,
551:                 "会话构建冲突复核（§7.2 两层防御第二层）: excludedPairs["
552:                     + std::to_string(excludedIt->second.first) + "]（理由'"
553:                     + excludedIt->second.second + "'）展开后覆盖 mandatoryPairs["
554:                     + std::to_string(mandatoryIndex) + "]（理由'"
555:                     + policy.collision.mandatoryPairs[mandatoryIndex].reason
556:                     + "'）的对象对（objA=" + pairKey.first.toCanonical()
557:                     + ", objB=" + pairKey.second.toCanonical() + "）——过滤不得隐藏必检");
558:         }
559:     }
560: }

568: CollisionEvaluationSession::CollisionEvaluationSession(
569:     const EngineeringPolicySet& policy, const CollisionScene& scene,
570:     const IPolicyNameContext& names, CollisionBackendDescriptor backend,
571:     std::shared_ptr<rw::proximity::CollisionStrategy> strategy,
572:     std::shared_ptr<std::mutex> backendQueryMutex)
573:     : m_policy(policy)
574:     , m_scene(scene)
575:     , m_nameMapIdentity(names.nameMapContentIdentity())
576:     , m_backend(std::move(backend))
577:     , m_strategy(std::move(strategy))
578:     , m_collisionEnabled(policy.collision.enabled)
579:     , m_device()                  // 评估半区（POL-T07——buildEvaluationHalf 填充）
580:     , m_distanceCapable(false)    // 默认无距离能力（探测后更新）
581:     , m_backendQueryMutex(std::move(backendQueryMutex))
582: {
584:     SceneObjectIndex objectIndex;
585:     validateSceneAndPolicy(m_policy, m_scene, names, m_strategy, objectIndex);
590:     std::map<PairKey, std::size_t> mandatoryExpanded;
591:     std::map<PairKey, std::pair<std::size_t, std::string>> excludedExpanded;
592:     expandRulesAndCheckConflicts(m_policy, objectIndex, mandatoryExpanded, excludedExpanded);
596:     std::set<PairKey> adjacentSet;
597:     for (const auto& adj : m_scene.adjacentLinkPairs) {
598:         adjacentSet.insert(detail::canonicalPair(adj.first, adj.second));
599:     }
602:     const std::set<CollisionDomain> enabledDomains(m_policy.collision.enabledDomains.begin(),
603:                                                    m_policy.collision.enabledDomains.end());
607:     std::vector<ScopedCollisionPair> expanded;
608:     for (std::size_t i = 0; i < m_scene.objects.size(); ++i) {
609:         for (std::size_t j = i + 1; j < m_scene.objects.size(); ++j) {
610:             const SceneObjectEntry& entryA = m_scene.objects[i];
611:             const SceneObjectEntry& entryB = m_scene.objects[j];
614:             const std::optional<CollisionDomain> domain =
615:                 detail::classifyDomainPair(entryA.role, entryB.role);
616:             if (!domain.has_value()) {
624:                 const PairKey droppedKey
625:                     = detail::canonicalPair(entryA.objectId, entryB.objectId);
626:                 if (mandatoryExpanded.find(droppedKey)
627:                     != mandatoryExpanded.end()) {
628:                     throw PolicyError(
629:                         PolicyErrorCode::PolicyObjectInvalid,
630:                         "policy/collision: mandatoryPairs 含矩阵未列组合（角色"
631:                         "对不入任何评估域——必检对被作用域定义隐藏，保守拒绝"
632:                         " F-808）");
633:                 }
634:                 continue;
635:             }
639:             if (!m_collisionEnabled || enabledDomains.find(*domain) == enabledDomains.end()) {
640:                 continue;
641:             }
645:             const PairKey key = detail::canonicalPair(entryA.objectId, entryB.objectId);
646:             ScopedCollisionPair pair;
647:             pair.objectA = key.first;
648:             pair.objectB = key.second;
650:             if (key.first == entryA.objectId) {
651:                 pair.roleA = entryA.role;
652:                 pair.roleB = entryB.role;
653:             }
654:             else {
655:                 pair.roleA = entryB.role;
656:                 pair.roleB = entryA.role;
657:             }
658:             pair.domain = *domain;
659:             const auto mandatoryIt = mandatoryExpanded.find(key);
660:             const auto excludedIt = excludedExpanded.find(key);
661:             if (mandatoryIt != mandatoryExpanded.end()) {
664:                 pair.status = ScopePairStatus::Mandatory;
665:                 pair.ruleIndex = mandatoryIt->second;
666:             }
667:             else if (excludedIt != excludedExpanded.end()) {
670:                 pair.status = ScopePairStatus::ExcludedByRule;
671:                 pair.ruleIndex = excludedIt->second.first;
672:                 pair.filterReason = excludedIt->second.second;
673:             }
674:             else if (adjacentSet.find(key) != adjacentSet.end()
675:                      && (*domain == CollisionDomain::Self
676:                          || *domain == CollisionDomain::Tool)
677:                      && m_policy.collision.excludeAdjacentLinksByDefault) {
681:                 pair.status = ScopePairStatus::ExcludedByAdjacencyDefault;
682:                 pair.ruleIndex = 0;
683:                 pair.filterReason = "default.adjacent-links";
684:             }
685:             else if (*domain == CollisionDomain::Scene) {
689:                 continue;
690:             }
691:             else {
694:                 pair.status = ScopePairStatus::InScopeDefault;
695:                 pair.ruleIndex = 0;
696:             }
697:             expanded.push_back(std::move(pair));
698:         }
699:     }
702:     std::sort(expanded.begin(), expanded.end(),
703:               [](const ScopedCollisionPair& l, const ScopedCollisionPair& r) {
704:                   if (l.objectA != r.objectA) {
705:                       return l.objectA < r.objectA;
706:                   }
707:                   return l.objectB < r.objectB;
708:               });
709:     m_scope.pairs = std::move(expanded);
717:     m_proximitySetup = std::make_shared<const rw::proximity::ProximitySetup>(
718:         buildPolicyProximitySetup(names));
721:     m_sessionIdentity = detail::computeSessionIdentity(m_policy.contentIdentity,
722:                                                        m_scene.sceneContentIdentity,
723:                                                        m_nameMapIdentity, m_backend);
728:     buildEvaluationHalf(names);
729: }
