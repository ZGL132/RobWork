818:     const auto byIdBytes = [](const core::ObjectId& a, const core::ObjectId& b) {
819:         return a.bytes < b.bytes;
820:     };
821:     applicability.modelObjects = normalizedSet(input.applicability.modelObjects, byIdBytes);
822:     applicability.taskObjects = normalizedSet(input.applicability.taskObjects, byIdBytes);
823:     applicability.caseObjects = normalizedSet(input.applicability.caseObjects, byIdBytes);
824: 
825:     // ⑥ 内容身份：PolicyCodec::contentIdentity——语义闭包 canonical 编码 →
826:     // SHA-256（CR-02 摘要单点；发布路径唯一调用——"只对通过校验并发布的
827:     // 策略计算"，§4.1）。
828:     const core::ContentIdentity identity = PolicyCodec::contentIdentity(
829:         input.schemaVersion, collision, jointThresholds, applicability,
830:         input.numericContractAnchor);
831: 
832:     // ⑦ 发布：EngineeringPolicySet::make（发布门只产出 Valid 实例——
833:     // 校验序①~⑦与其门内核对互补：此处入参已满足全部前置，门校验若因
834:     // 逻辑缺陷触发异常将 fail-fast 传播——不吞错）。
835:     // 就地构造（emplace）：EngineeringPolicySet 全成员 const——拷贝/移动
836:     // 赋值被编译期删除（§4.6 不可变性的类型层强制），optional 的赋值算子
837:     // 相应不可用；构造途径（拷贝/移动构造）不受影响。
838:     outcome.policy.emplace(EngineeringPolicySet::make(
839:         input.policyObject, input.schemaVersion, identity, std::move(collision),
840:         std::move(jointThresholds), std::move(applicability), input.origin,
841:         PolicyValidationState::Valid, outcome.diagnostics,   // 全量诊断随发布对象附带（此时仅 Info）
842:         input.compatibilityNotes, input.numericContractAnchor));
843:     return outcome;
844: }
