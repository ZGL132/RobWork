653:     // =================================================================
654:     {
655:         // 对象品类：元数据对象→域对象（计划序）。共享判定（目标已存在）
656:         // 复用 ObjectStore 读回校验通道——摘要不符抛 StoreCorrupt，与
657:         // publishObject 的共享语义同源同码（实现口径登记见头文件）。
658:         const fs::path stagedMetaPath =
659:             txDir / "objects" / metadataOid.toCanonical() / cvHex(metadataCv);
660:         publishOneObject(stagedMetaPath, metadataOid, metadataCv);
661:         for (const StagedObject& s : staged) {
662:             publishOneObject(
663:                 txDir / "objects" / s.oid.toCanonical() / cvHex(s.cv),
664:                 s.oid, s.cv);
665:         }
666:         // 修订目录：先建目录，command.json 次之，manifest.json 最后
667:         // （§7.1"对象先行、修订清单最后"）。
668:         const fs::path targetRevDir = revisionDir(newRevId);
669:         const win32::FileResult dirCr =
670:             m_fileOps->createDirectories(targetRevDir.wstring());
671:         if (!dirCr.ok) {
672:             throw osToStoreError(dirCr.osError, "publish-mkdir",
673:                                  targetRevDir.string());
674:         }
675:         const win32::FileResult cmdPub = m_fileOps->publishNew(
676:             stagedCommand.wstring(),
677:             (targetRevDir / "command.json").wstring());
678:         if (!cmdPub.ok) {
679:             throw osToStoreError(cmdPub.osError, "publish-command",
680:                                  (targetRevDir / "command.json").string());
681:         }
682:         const win32::FileResult manPub = m_fileOps->publishNew(
683:             stagedManifest.wstring(),
684:             (targetRevDir / "manifest.json").wstring());
685:         if (!manPub.ok) {
686:             throw osToStoreError(manPub.osError, "publish-manifest",
687:                                  (targetRevDir / "manifest.json").string());
688:         }
689:     }
690: 
691:     // =================================================================
692:     // 第 5 步：提交指针切换（§7.1"写 HEAD.new（完整 HeadRecord）→ flush →
693:     // MoveFileExW(HEAD.new→HEAD, REPLACE_EXISTING|WRITE_THROUGH)——唯一
694:     // 提交点；切换后重读 HEAD 自校验 manifestDigest"）。
695:     // =================================================================
696:     HeadRecord newHead;
697:     newHead.schemaVersion = currentHead.schemaVersion;
698:     newHead.projectId = currentHead.projectId;
699:     newHead.revisionId = newRevId;
700:     newHead.revisionSeq = revSeq;
701:     newHead.branchId = plan.branchId;
702:     newHead.manifestDigest = manifestDigestHex;
703:     const fs::path stagedHead = txDir / "head.new";
704:     {
705:         const std::string headBytes = codec::dump(newHead);
706:         const win32::FileResult wr =
707:             file.writeThrough(stagedHead.wstring(), headBytes.data(),
708:                               headBytes.size());
709:         if (!wr.ok) {
710:             throw osToStoreError(wr.osError, "stage-head",
711:                                  stagedHead.string());
712:         }
713:     }
714:     {
715:         const fs::path headPath = m_projectDir / "HEAD";
716:         // replaceExisting＝MoveFileExW(REPLACE_EXISTING|WRITE_THROUGH)（§7.2
717:         // 行 3）。本调用成功即"唯一提交点"成立——此后本次修订视为已提交，
718:         // 后续任何失败（第 6/7 步）不回滚（D-18）。失败→环境码；内容已
719:         // 发布但 HEAD 未变＝第 4 步残留语义（闭包外可识别）。
720:         const win32::FileResult rep =
721:             m_fileOps->replaceExisting(stagedHead.wstring(),
722:                                        headPath.wstring());
723:         if (!rep.ok) {
724:             throw osToStoreError(rep.osError, "head-switch", headPath.string());
725:         }
726:         // 唯一提交点已过——解除幻影守卫（本修订已是提交事实：此后自校验/
727:         // 事件/清理失败均走 D-18 不回滚语义，索引注册必须保留）。
728:         phantomGuard.armed = false;
729:         // 自校验（§7.1 第 5 步"切换后重读 HEAD 自校验 manifestDigest"）：
730:         // 重读+重解与内存事实逐字段比对。失败＝提交点内容异常的极端态
731:         // （读失败/内容不符）——D-18 不回滚：抛 StoreCorrupt 硬错误，恢复
732:         // 扫描④以磁盘 HEAD 实际内容为准兜底（§7.1"崩溃于切换前后以 HEAD
733:         // 实际内容为准（字节级判定）"）。
734:         const auto headBytes = readAllBytes(headPath);
735:         bool selfCheckOk = headBytes.has_value();
736:         if (selfCheckOk) {
737:             try {
738:                 selfCheckOk = (codec::parseHeadRecord(*headBytes) == newHead);
739:             } catch (const StoreError&) {
740:                 selfCheckOk = false;  // 重读内容解析失败＝自校验不过
741:             }
742:         }
743:         if (!selfCheckOk) {
744:             reportDev("head-selfcheck-failed（HEAD 切换后重读不符——提交点"
745:                       "异常，恢复扫描④兜底） rev=" + newRevId.toCanonical());
746:             throw StoreError(StoreErrorCode::StoreCorrupt,
747:                              "project/tx-engine: head-selfcheck-failed path="
748:                                  + (m_projectDir / "HEAD").string());
749:         }
750:     }
751: 
752:     // =================================================================
753:     // 第 6 步：事件发布（§7.1"bus.publish(RevisionCommitted→
754:     // DependencyInvalidated)（FIFO；失败重试一次＋开发诊断，不回滚——D-18）"）。
755:     // 总线可空＝跳过（§5.1 装配口径）。失败不影响提交事实。
756:     // =================================================================
757:     if (m_eventBus != nullptr) {
758:         core::RevisionCommittedPayload committed;
759:         committed.project = currentHead.projectId;
760:         committed.branch = plan.branchId;
761:         committed.revision = newRevId;
762:         committed.parent = plan.parentRevisionId;
763:         publishEventWithRetry(core::DomainEvent::make(committed),
764:                               "revision-committed");
765:         // FIFO：失效通知严格在提交事件之后发布（订阅方先刷新视图再失效
766:         // 依赖——§7.1 第 6 步"RevisionCommitted→DependencyInvalidated"序）。
767:         core::DependencyInvalidatedPayload invalidated;
768:         invalidated.project = currentHead.projectId;
769:         invalidated.branch = plan.branchId;
770:         invalidated.revision = newRevId;
771:         publishEventWithRetry(core::DomainEvent::make(invalidated),
772:                               "dependency-invalidated");
773:     }
774: 
775:     // =================================================================
776:     // 第 7 步：清理（§7.1"删除 .staging/<tx-id>/（含 tmp）；清理失败仅
777:     // 开发诊断"——已提交状态不受影响，残留不参与任何读取路径）。
778:     // =================================================================
779:     {
780:         const win32::FileResult rm = m_fileOps->removeTree(txDir.wstring());
781:         if (!rm.ok) {
782:             reportDev("cleanup-failed（.staging 清理失败——已提交状态不受"
783:                       "影响，残留由下次启动扫描处置，F7 边界） dir="
784:                           + txDir.string() + " osError="
785:                           + std::to_string(rm.osError));
786:         }
787:     }
788: 
789:     // ---- 提交结果（供命令服务组装返回值/事件消费，PRJ-T10） ----
790:     CommitResult result;
791:     result.revisionId = newRevId;
792:     result.revisionSeq = revSeq;
793:     result.metadataRef = candidateRef;
794:     result.manifestDigest = manifestDigestHex;
795:     result.newHead = newHead;
796:     return result;
797: }
