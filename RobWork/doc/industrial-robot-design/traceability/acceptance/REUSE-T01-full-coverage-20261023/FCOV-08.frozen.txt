514:     // 必检规则展开（承载序遍历；map 首条准入——重复对确定性吸收）。

526:         }
527:     }

530:         const PairRule& rule = policy.collision.excludedPairs[i];
531:         const std::vector<core::ObjectId> first = expandRuleTarget(rule.first, objectIndex);
532:         const std::vector<core::ObjectId> second = expandRuleTarget(rule.second, objectIndex);
533:         for (const core::ObjectId& a : first) {
534:             for (const core::ObjectId& b : second) {
535:                 if (a == b) {
536:                     continue;
537:                 }
538:                 excludedExpanded.emplace(detail::canonicalPair(a, b),
539:                                          std::make_pair(i, rule.reason));
540:             }
541:         }
542:     }
543:     // 冲突同码复核（§7.2："排除∩必检=∅ 必须成立……违反→POLICY-RULE-

546:     for (const auto& [pairKey, mandatoryIndex] : mandatoryExpanded) {
547:         const auto excludedIt = excludedExpanded.find(pairKey);
548:         if (excludedIt != excludedExpanded.end()) {

558:         }
